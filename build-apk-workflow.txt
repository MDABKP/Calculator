name: Build APK
on:
  workflow_dispatch:
  push:
    branches: [main]
permissions:
  contents: write
jobs:
  build:
    runs-on: ubuntu-latest
    env:
      KEYSTORE_BASE64: ${{ secrets.KEYSTORE_BASE64 }}
      KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
      KEY_ALIAS: ${{ secrets.KEY_ALIAS }}
      KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - run: npm install
      - run: npx cap add android
      - run: npx capacitor-assets generate --android --iconBackgroundColor '#16171a' --iconBackgroundColorDark '#16171a' --splashBackgroundColor '#16171a' --splashBackgroundColorDark '#16171a'
      - name: Install custom icons
        run: |
          find android/app/src/main/res -name 'ic_launcher*.webp' -delete
          cp -r icons/res/* android/app/src/main/res/
      - name: Tweak colors and orientation
        run: |
          sed -i 's/#3F51B5/#16171A/g; s/#303F9F/#16171A/g' android/app/src/main/res/values/colors.xml || true
          sed -i 's/android:configChanges/android:screenOrientation="portrait" android:configChanges/' android/app/src/main/AndroidManifest.xml || true
      - name: Ensure INTERNET and VIBRATE permissions
        run: |
          M=android/app/src/main/AndroidManifest.xml
          for P in INTERNET VIBRATE; do
            if ! grep -q "android.permission.$P" "$M"; then
              sed -i "s#<application#<uses-permission android:name=\"android.permission.$P\" />\n    <application#" "$M"
            fi
          done
          grep -n "uses-permission" "$M"
      - name: Compute version automatically
        run: |
          # Version = 1.<build number>, e.g. build 1 -> 1.1, build 12 -> 1.12
          VER="1.${{ github.run_number }}"
          echo "Version $VER build ${{ github.run_number }}"
          sed -i "s/APP_VERSION=\"[0-9.]*\"/APP_VERSION=\"$VER\"/" www/index.html
          grep -q "APP_VERSION=\"$VER\"" www/index.html || { echo "APP_VERSION not found in www/index.html"; exit 1; }
          echo "VER=$VER" >> "$GITHUB_ENV"
      - run: npx cap sync android
      - name: Set version
        run: |
          echo "Version $VER build ${{ github.run_number }}"
          sed -i 's/versionCode [0-9]*/versionCode ${{ github.run_number }}/' android/app/build.gradle
          sed -i "s/versionName \"[^\"]*\"/versionName \"$VER\"/" android/app/build.gradle
      - name: Prepare signing key from GitHub Secrets
        run: |
          for v in KEYSTORE_BASE64 KEYSTORE_PASSWORD KEY_ALIAS KEY_PASSWORD; do
            if [ -z "${!v}" ]; then
              echo "Missing GitHub Secret: $v (Settings > Secrets and variables > Actions)"
              exit 1
            fi
          done
          echo "$KEYSTORE_BASE64" | base64 -d > "$RUNNER_TEMP/release.keystore"
          test -s "$RUNNER_TEMP/release.keystore" || { echo "KEYSTORE_BASE64 is empty or invalid"; exit 1; }
          echo "KEYSTORE_PATH=$RUNNER_TEMP/release.keystore" >> "$GITHUB_ENV"
      - name: Wire signing into Gradle (values come from environment, nothing hardcoded)
        run: |
          cat >> android/app/build.gradle <<'GRADLE'

          android {
              signingConfigs {
                  debug {
                      storeFile file(System.getenv("KEYSTORE_PATH"))
                      storePassword System.getenv("KEYSTORE_PASSWORD")
                      keyAlias System.getenv("KEY_ALIAS")
                      keyPassword System.getenv("KEY_PASSWORD")
                  }
                  release {
                      storeFile file(System.getenv("KEYSTORE_PATH"))
                      storePassword System.getenv("KEYSTORE_PASSWORD")
                      keyAlias System.getenv("KEY_ALIAS")
                      keyPassword System.getenv("KEY_PASSWORD")
                  }
              }
              buildTypes {
                  release {
                      signingConfig signingConfigs.release
                  }
              }
              lint {
                  checkReleaseBuilds false
                  abortOnError false
              }
          }
          GRADLE
      - run: chmod +x android/gradlew && cd android && ./gradlew assembleDebug
      - name: Verify debug APK is signed with your key
        run: |
          APK=android/app/build/outputs/apk/debug/app-debug.apk
          APKSIGNER=$(ls $ANDROID_HOME/build-tools/*/apksigner 2>/dev/null | sort -V | tail -1)
          if [ -z "$APKSIGNER" ]; then echo "apksigner not found"; exit 1; fi
          EXPECTED=$(keytool -list -v -keystore "$KEYSTORE_PATH" -alias "$KEY_ALIAS" -storepass "$KEYSTORE_PASSWORD" 2>/dev/null | grep 'SHA256:' | head -1 | sed 's/.*SHA256: *//' | tr -d ':' | tr 'A-Z' 'a-z')
          getfp() { "$APKSIGNER" verify --print-certs "$APK" | grep 'SHA-256 digest' | head -1 | sed 's/.*: *//' | tr -d ': ' | tr 'A-Z' 'a-z'; }
          ACTUAL=$(getfp)
          echo "your key: $EXPECTED"
          echo "apk key:  $ACTUAL"
          [ -n "$EXPECTED" ] && [ "$EXPECTED" = "$ACTUAL" ] || { echo "SIGNATURE MISMATCH: debug APK is not signed with your key"; exit 1; }
      - name: Build release APK (optional - the debug APK above stays as fallback)
        continue-on-error: true
        run: |
          cd android
          ./gradlew assembleRelease > release-build.log 2>&1 || { echo "=== RELEASE BUILD FAILED - last 40 lines ==="; tail -n 40 release-build.log; exit 1; }
          cd ..
          APK=android/app/build/outputs/apk/release/app-release.apk
          test -f "$APK" || { echo "release APK was not produced"; exit 1; }
          APKSIGNER=$(ls $ANDROID_HOME/build-tools/*/apksigner 2>/dev/null | sort -V | tail -1)
          EXPECTED=$(keytool -list -v -keystore "$KEYSTORE_PATH" -alias "$KEY_ALIAS" -storepass "$KEYSTORE_PASSWORD" 2>/dev/null | grep 'SHA256:' | head -1 | sed 's/.*SHA256: *//' | tr -d ':' | tr 'A-Z' 'a-z')
          getfp() { "$APKSIGNER" verify --print-certs "$APK" | grep 'SHA-256 digest' | head -1 | sed 's/.*: *//' | tr -d ': ' | tr 'A-Z' 'a-z'; }
          ACTUAL=$(getfp)
          echo "your key:    $EXPECTED"
          echo "release key: $ACTUAL"
          if [ -z "$EXPECTED" ] || [ "$EXPECTED" != "$ACTUAL" ]; then
            echo "release APK is not signed with your key - removing it, debug APK stays"
            rm -f "$APK"
            exit 1
          fi
      - name: Rename APKs
        run: |
          mkdir -p dist
          D=android/app/build/outputs/apk/debug/app-debug.apk
          R=android/app/build/outputs/apk/release/app-release.apk
          if [ -f "$D" ]; then cp "$D" "dist/Sohoj-Calculator-v${VER}-debug.apk"; fi
          if [ -f "$R" ]; then cp "$R" "dist/Sohoj-Calculator-v${VER}-release.apk"; fi
          ls -la dist
      - name: Remove keystore from runner
        if: always()
        run: rm -f "$RUNNER_TEMP/release.keystore"
      - uses: actions/upload-artifact@v4
        with:
          name: calculator-apk
          path: dist/*-debug.apk
      - uses: actions/upload-artifact@v4
        with:
          name: calculator-apk-release
          path: dist/*-release.apk
          if-no-files-found: ignore
      - uses: softprops/action-gh-release@v2
        with:
          tag_name: v${{ github.run_number }}
          name: Sohoj Calculator v${{ env.VER }}
          body: |
            📲 কোনটা নামাবেন?
            • Sohoj-Calculator-v${{ env.VER }}-release.apk: এটাই নামান। দ্রুত চলে ও প্রস্তাবিত।
            • Sohoj-Calculator-v${{ env.VER }}-debug.apk: শুধু উপরেরটা না চললে ব্যবহার করুন।

            ⚙️ ইনস্টলের নিয়ম
            1. APK ফাইলে চাপ দিয়ে ইনস্টল করুন।
            2. "অজানা উৎস" বা Play Protect-এর সতর্কতা এলে "তবুও ইনস্টল করুন" বেছে নিন।
               (অ্যাপটা Play Store-এ নেই, তাই এই সতর্কতা স্বাভাবিক।)

            🔄 আপডেট করতে নতুন APK নামিয়ে ইনস্টল করুন, আগেরটা মুছতে হবে না।

            শুধু এই রিপোর Releases পাতা থেকেই APK নামান।
          files: |
            dist/*-release.apk
            dist/*-debug.apk
